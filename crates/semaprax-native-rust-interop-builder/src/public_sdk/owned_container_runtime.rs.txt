#[repr(C)]
#[derive(Clone, Copy, Default)]
pub struct SpxContainer { tag:u8, reserved:[u8;7], error:i64, payload:SpxOwner }
type NativeValue = @NATIVE_TYPE@;
const ACTIVE_TAG:u8=@ACTIVE_TAG@;
const OPTION:bool=@OPTION@;
fn zero_owner(owner:SpxOwner)->bool {owner.context==0 && owner.generation==0 && owner.slot==0}
#[no_mangle]
pub extern "C" fn spx_container_validate(context:u64, value:SpxContainer)->i32 {
    if value.tag>1 || value.reserved!=[0;7] {return 3;}
    if value.tag==ACTIVE_TAG { if value.error!=0 {return 3;} }
    else if !zero_owner(value.payload) || (OPTION && value.error!=0) {return 3;}
    CONTEXTS.with(|contexts| {
        let Ok(contexts)=contexts.try_borrow() else {return 6;};
        let Some(table)=contexts.iter().find(|c|c.id==context) else {return 3;};
        if value.tag!=ACTIVE_TAG {return 0;}
        if value.payload.context!=context {return 3;}
        let Ok(index)=usize::try_from(value.payload.slot) else {return 3;};
        let Some(slot)=table.slots.get(index) else {return 3;};
        if slot.generation!=value.payload.generation || slot.value.is_none() {return 3;}
        0
    })
}
#[no_mangle]
pub unsafe extern "C" fn spx_container_new(context:u64,arg:i64,out:*mut SpxContainer)->i32 {
    if out.is_null() || (out as usize)%std::mem::align_of::<SpxContainer>()!=0 {return 3;}
    let produced=caught(||CONTEXTS.with(|contexts|{
        let mut contexts=contexts.try_borrow_mut().map_err(|_|6)?;
        let context=contexts.iter_mut().find(|c|c.id==context).ok_or(3)?;
        let free=context.slots.iter().position(|s|s.value.is_none() && s.generation!=u64::MAX);
        if free.is_none() {
            if context.slots.len()>=32 {return Err(4);}
            context.slots.try_reserve(1).map_err(|_|4)?;
        }
        let constructor:fn(i64)->NativeValue=@CONSTRUCTOR@;
        let produced=constructor(arg);
        let (tag,error,payload)=match produced { @DECODE@ };
        let mut wire=SpxContainer{tag,reserved:[0;7],error,payload:SpxOwner::default()};
        if let Some(value)=payload {
            if value.len()>4096 || value.capacity()>4096 {return Err(4);}
            let index=if let Some(index)=free {index} else {context.slots.push(Slot{generation:0,value:None});context.slots.len()-1};
            let slot=&mut context.slots[index];slot.generation+=1;slot.value=Some(value);
            wire.payload=SpxOwner{context:context.id,generation:slot.generation,slot:index as u64};
        }
        Ok(wire)
    }));
    match produced {
        // The C caller grants an aligned writable out-slot. No Rust layout or
        // allocation pointer crosses it; all scalar fields are initialized.
        Ok(Ok(value))=>{unsafe{out.write(value)};0},Ok(Err(s))|Err(s)=>s
    }
}
#[no_mangle]
pub unsafe extern "C" fn spx_container_consume(context:u64,wire:SpxContainer,arg:i64,out:*mut u8)->i32 {
    if out.is_null(){return 3;}
    let status=spx_container_validate(context,wire);if status!=0{return status;}
    let value:NativeValue=if wire.tag==ACTIVE_TAG {
        let payload=match take(context,wire.payload){Ok(v)=>v,Err(s)=>return s};
        @ACTIVE_VALUE@
    } else {@INACTIVE_VALUE@};
    let consumer:fn(NativeValue,i64)->bool=@CONSUMER@;
    match caught(||consumer(value,arg)) {
        // The caller grants a writable byte; failure never publishes it.
        Ok(value)=>{unsafe{out.write(u8::from(value))};0},Err(s)=>s
    }
}
#[no_mangle]
pub extern "C" fn spx_container_drop(context:u64,wire:SpxContainer)->i32 {
    let status=spx_container_validate(context,wire);if status!=0{return status;}
    if wire.tag==ACTIVE_TAG {spx_payload_drop(context,wire.payload)}else{0}
}

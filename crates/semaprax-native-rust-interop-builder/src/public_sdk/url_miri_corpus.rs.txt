#[test]
fn carrier_and_loan_corpus() {
    assert_carrier_mutation_corpus();
    let context=spx_result_owner_context_new();
    let text=b"https://example.invalid/miri";
    let mut result=SpxOwnerResult::default();
    assert_eq!(unsafe {spx_result_owner_new_utf8(context,text.as_ptr(),text.len() as u64,&mut result)},0);
    let owner=result.owner;
    let mut view=SpxStrView::default();
    assert_eq!(unsafe {spx_url_owner_view(context,owner,&mut view)},0);
    // Read the authentic live backing, then revalidate while the Rust slice
    // remains live. Miri checks actual pointer provenance and aliasing here.
    let bytes=unsafe {std::slice::from_raw_parts(view.data,view.length as usize)};
    assert_eq!(bytes,text);
    let mut length=0;
    assert_eq!(unsafe {spx_url_view_length(context,view,&mut length)},0);
    assert_eq!(with_exclusive_url(context,owner,|_|()).err(),Some(6));
    assert_eq!(bytes,text);
    assert_eq!(spx_url_view_end(context,view),0);
    with_exclusive_url(context,owner,|borrowed| {
        let before=target_call_count();
        assert_eq!(unsafe {spx_url_owner_view(context,owner,&mut view)},6);
        assert_eq!(unsafe {spx_result_owner_drop(context,owner)},6);
        assert_eq!(with_exclusive_url(context,owner,|_|()).err(),Some(6));
        assert_eq!(target_call_count(),before);
        borrowed.set_path("/miri-mutated");
    }).unwrap();
    assert_eq!(unsafe {spx_url_owner_view(context,owner,&mut view)},0);
    let bytes=unsafe {std::slice::from_raw_parts(view.data,view.length as usize)};
    assert_eq!(bytes,b"https://example.invalid/miri-mutated");
    assert_eq!(spx_url_view_end(context,view),0);
    assert_eq!(unsafe {spx_result_owner_drop(context,owner)},0);
    assert_eq!(spx_result_owner_context_close(context),0);
    assert_eq!((live_owner_count(),live_view_count()),(0,0));
}

// Test-only injected controls; absent from the public package renderer.
pub fn assert_view_controls() {
    let context = spx_result_owner_context_new(); assert_ne!(context, 0);
    let mut result = SpxOwnerResult::default();
    let text = b"https://example.invalid/path";
    assert_eq!(unsafe { spx_result_owner_new_utf8(context,text.as_ptr(),text.len() as u64,&mut result) },0);
    let owner = result.owner; assert_eq!(result.tag,1);
    let mut view = SpxStrView::default();
    assert_eq!(unsafe {spx_url_owner_view(context,owner,&mut view)},0);
    assert_eq!(view.data as usize,target_view_pointer());
    let calls = target_call_count();
    let mut second = SpxStrView::default();
    assert_eq!(unsafe {spx_url_owner_view(context,owner,&mut second)},6);
    assert_eq!(unsafe {spx_result_owner_drop(context,owner)},6);
    assert_eq!(spx_result_owner_context_close(context),5);
    assert_eq!(target_call_count(),calls,"refusal must precede target entry");
    let mut length=999;
    assert_eq!(unsafe {spx_url_view_length(context,view,&mut length)},0);assert_eq!(length,text.len() as u64);
    for forged in [SpxStrView { lease:view.lease+1,..view },SpxStrView { data:std::ptr::null(),..view },
        SpxStrView { owner:SpxOwner { generation:owner.generation+1,..owner },..view },
        SpxStrView { owner:SpxOwner { context:owner.context+1,..owner },..view },
        SpxStrView { length:view.length+1,..view }] {
        length=999; assert_eq!(unsafe {spx_url_view_length(context,forged,&mut length)},3);assert_eq!(length,999);
        assert_eq!(spx_url_view_end(context,forged),3);
    }
    CONTEXTS.with(|contexts| {
        let _held=contexts.borrow_mut();
        assert_eq!(unsafe {spx_url_owner_view(context,owner,&mut second)},6);
    });
    assert_eq!(target_call_count(),calls,"re-entry refused before target");
    assert_eq!(spx_url_view_end(context,view),0);
    assert_eq!(spx_url_view_end(context,view),3);
    assert_eq!(unsafe {spx_result_owner_drop(context,owner)},0);
    assert_eq!(unsafe {spx_url_view_length(context,view,&mut length)},3);
    assert_eq!(spx_result_owner_context_close(context),0);
    assert_eq!(unsafe {spx_url_owner_view(context,owner,&mut second)},3);
    assert_eq!(live_owner_count(),0);assert_eq!(live_view_count(),0);
}

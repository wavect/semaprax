RpcId: TypeAlias = int | str
class ResultEnvelope(TypedDict):
    schema: str
    protocol: str
    image_revision: str
    project_revision: str
    payload: Any
class RpcDiagnosticLocation(TypedDict):
    line: int
    column: int
    start: int
    end: int
class RpcDiagnostic(TypedDict):
    code: str
    severity: Literal['error', 'warning']
    message: str
    path: str | None
    location: RpcDiagnosticLocation | None
    help: str | None
class RpcDiagnosticData(TypedDict):
    schema: Literal['semaprax.image-agent-application-error-data.v1']
    diagnostics: list[RpcDiagnostic]
class RpcError(TypedDict):
    code: int
    message: str
class TypedRpcError(RpcError):
    data: NotRequired[RpcDiagnosticData]
class RpcResponseError(ValueError):
    def __init__(self, error: TypedRpcError):
        super().__init__(f"RPC {error['code']}: {error['message']}")
        self.error = error

def _same(left: Any, right: Any) -> bool:
    if type(left) is not type(right):
        return False
    if isinstance(left, dict):
        return left.keys() == right.keys() and all(_same(left[key], right[key]) for key in left)
    if isinstance(left, list):
        return len(left) == len(right) and all(_same(a, b) for a, b in zip(left, right))
    return left == right

class _ShapeLimitError(ValueError):
    pass

def _check(value: Any, schema: dict[str, Any], depth: int = 0, budget: list[int] | None = None) -> None:
    if budget is None:
        budget = [1048576]
    budget[0] -= 1
    if budget[0] < 0:
        raise _ShapeLimitError('shape work exceeded')
    if depth > 128:
        raise _ShapeLimitError('shape depth exceeded')
    if '$ref' in schema:
        reference = schema['$ref']
        if reference in META['documents']:
            return _check(value, META['documents'][reference], depth + 1, budget)
        if reference in META['unbundled'] and isinstance(value, dict) and value.get('schema') == reference[4:]:
            return
        raise ValueError('unresolved or foreign payload schema')
    for alternative in ('oneOf', 'anyOf'):
        if alternative in schema:
            count = 0
            for branch in schema[alternative]:
                try:
                    _check(value, branch, depth + 1, budget)
                    count += 1
                except _ShapeLimitError:
                    raise
                except ValueError:
                    pass
            if (count != 1 if alternative == 'oneOf' else count == 0):
                raise ValueError('shape alternatives mismatch')
    if 'const' in schema and not _same(value, schema['const']):
        raise ValueError('constant mismatch')
    if 'enum' in schema and not any(_same(value, item) for item in schema['enum']):
        raise ValueError('choice mismatch')
    if 'not' in schema and any(_same(value, item) for item in schema['not']['enum']):
        raise ValueError('excluded choice')
    kind = schema.get('type')
    if kind == 'null' and value is not None:
        raise ValueError('null required')
    if kind == 'boolean' and type(value) is not bool:
        raise ValueError('boolean required')
    if kind == 'integer' and (type(value) is not int or value < schema.get('minimum', -(2**63)) or value > schema.get('maximum', 2**64-1)):
        raise ValueError('integer bound')
    if kind == 'string':
        if not isinstance(value, str) or len(value) < schema.get('minLength', 0) or len(value) > schema.get('maxLength', 2**64) or len(value.encode('utf-8')) > schema.get('x-max-utf8-bytes', 2**64) or ('pattern' in schema and re.fullmatch(schema['pattern'], value) is None):
            raise ValueError('string bound/format')
    if kind == 'object':
        if not isinstance(value, dict):
            raise ValueError('object required')
        if any(key not in value for key in schema.get('required', [])):
            raise ValueError('missing field')
        # Reject a foreign discriminant before walking a recursive body/argument.
        for key, field in schema.get('properties', {}).items():
            if key in value and 'const' in field:
                _check(value[key], field, depth + 1, budget)
        for key, item in value.items():
            if key in schema.get('properties', {}):
                _check(item, schema['properties'][key], depth + 1, budget)
            elif schema.get('additionalProperties') is False:
                raise ValueError('unknown field: ' + key)
    if kind == 'array':
        if not isinstance(value, list) or len(value) < schema.get('minItems', 0) or len(value) > schema.get('maxItems', 2**64):
            raise ValueError('array bound')
        if 'items' in schema:
            for item in value:
                _check(item, schema['items'], depth + 1, budget)
        if schema.get('uniqueItems'):
            seen: set[str] = set()
            for item in value:
                budget[0] -= 1
                if budget[0] < 0:
                    raise _ShapeLimitError('shape work exceeded')
                if not isinstance(item, str) or item in seen:
                    raise ValueError('unique string items required')
                seen.add(item)

def _id(request_id: RpcId) -> None:
    if type(request_id) is int:
        if request_id < 0 or request_id > 2**64-1:
            raise ValueError('request id bound')
    else:
        _check(request_id, {'type':'string', 'minLength':1, 'x-max-utf8-bytes':128, 'pattern':r'^[^\u0000-\u001f\u007f-\u009f]+$'})

def request(request_id: RpcId, method: str, params: dict[str, Any]) -> str:
    _id(request_id)
    if method not in META['methods']:
        raise ValueError('method unavailable')
    spec = json.loads(json.dumps(META['methods'][method]['params']))
    for field in spec['properties'].values():
        field.pop('$ref', None)  # Nested semantic constructors remain compiler-admitted.
    _check(params, spec)
    line = json.dumps(dict(jsonrpc='2.0', id=request_id, method=method, params=params), ensure_ascii=False, allow_nan=False, separators=(',', ':'))
    if len(line.encode('utf-8')) > 65536:
        raise ValueError('request byte bound')
    return line + '\n'

def _pairs(pairs: list[tuple[str, Any]]) -> dict[str, Any]:
    result: dict[str, Any] = {}
    for key, value in pairs:
        if key in result:
            raise ValueError('duplicate response key')
        result[key] = value
    return result

def decode(line: str, method: str, request_id: RpcId) -> ResultEnvelope:
    _id(request_id)
    if method not in META['methods'] or len(line.encode('utf-8')) > 1048576:
        raise ValueError('response bound/method')
    def invalid_number(value: str) -> Any:
        raise ValueError('non-JSON number: ' + value)
    value = json.loads(line, object_pairs_hook=_pairs, parse_constant=invalid_number)
    if not isinstance(value, dict) or value.get('jsonrpc') != '2.0':
        raise ValueError('response envelope')
    if 'error' in value:
        raise ValueError(_typed_rpc_error(value, request_id))
    digest = {'type':'string','pattern':r'^sha256:[0-9a-f]{64}$'}
    _check(value, {'type':'object','required':['jsonrpc','id','result'],'additionalProperties':False,'properties':{'jsonrpc':{'const':'2.0'},'id':{'const':request_id},'result':{'type':'object','required':['schema','protocol','image_revision','project_revision','payload'],'additionalProperties':False,'properties':{'schema':{'const':RESULT_SCHEMA},'protocol':{'const':PROTOCOL},'image_revision':digest,'project_revision':digest,'payload':META['methods'][method]['payload']}}}})
    return value['result']

_DIAGNOSTIC_LOCATION = {'type':'object','required':['line','column','start','end'],'additionalProperties':False,'properties':{'line':{'type':'integer','minimum':0},'column':{'type':'integer','minimum':0},'start':{'type':'integer','minimum':0},'end':{'type':'integer','minimum':0}}}
_DIAGNOSTIC_ROW = {'type':'object','required':['code','severity','message','path','location','help'],'additionalProperties':False,'properties':{'code':{'type':'string'},'severity':{'enum':['error','warning']},'message':{'type':'string'},'path':{'oneOf':[{'type':'string'},{'type':'null'}]},'location':{'oneOf':[_DIAGNOSTIC_LOCATION,{'type':'null'}]},'help':{'oneOf':[{'type':'string'},{'type':'null'}]}}}
_TYPED_RPC_ERROR = {'type':'object','required':['code','message'],'additionalProperties':False,'properties':{'code':{'type':'integer'},'message':{'type':'string'},'data':{'type':'object','required':['schema','diagnostics'],'additionalProperties':False,'properties':{'schema':{'const':'semaprax.image-agent-application-error-data.v1'},'diagnostics':{'type':'array','minItems':1,'items':_DIAGNOSTIC_ROW}}}}}

def _typed_rpc_error(value: dict[str, Any], request_id: RpcId) -> TypedRpcError:
    _check(value, {'type':'object','required':['jsonrpc','id','error'],'additionalProperties':False,'properties':{'jsonrpc':{'const':'2.0'},'id':{},'error':_TYPED_RPC_ERROR}})
    error = value['error']
    if 'data' in error and (error['code'] != -32000 or value['id'] is None or value['id'] != request_id or type(value['id']) is not type(request_id)):
        raise ValueError('application error data requires correlated -32000 response')
    if value['id'] is not None and (value['id'] != request_id or type(value['id']) is not type(request_id)):
        raise ValueError('response id mismatch')
    return error

def decode_typed(line: str, method: str, request_id: RpcId) -> ResultEnvelope:
    """Decode a selected response while preserving structured application diagnostics."""
    _id(request_id)
    if method not in META['methods'] or len(line.encode('utf-8')) > 1048576:
        raise ValueError('response bound/method')
    def invalid_number(value: str) -> Any:
        raise ValueError('non-JSON number: ' + value)
    value = json.loads(line, object_pairs_hook=_pairs, parse_constant=invalid_number)
    if not isinstance(value, dict) or value.get('jsonrpc') != '2.0':
        raise ValueError('response envelope')
    if 'error' not in value:
        return decode(line, method, request_id)
    raise RpcResponseError(_typed_rpc_error(value, request_id))

#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
#[serde(untagged)]
pub enum RpcId { Number(u64), Text(String) }
#[derive(Clone, Debug, Deserialize)]
#[serde(deny_unknown_fields)]
pub struct ResultEnvelope {
    pub schema: String, pub protocol: String, pub image_revision: String,
    pub project_revision: String, pub payload: Value,
}
#[derive(Clone, Debug, Deserialize)]
#[serde(deny_unknown_fields)]
pub struct RpcError { pub code: i64, pub message: String }
#[derive(Clone, Debug, PartialEq, Deserialize)]
#[serde(deny_unknown_fields)]
pub struct RpcDiagnosticLocation { pub line: u64, pub column: u64, pub start: u64, pub end: u64 }
#[derive(Clone, Debug, PartialEq, Deserialize)]
#[serde(deny_unknown_fields)]
pub struct RpcDiagnostic { pub code: String, pub severity: RpcDiagnosticSeverity, pub message: String, pub path: Option<String>, pub location: Option<RpcDiagnosticLocation>, pub help: Option<String> }
#[derive(Clone, Debug, PartialEq, Deserialize)]
#[serde(rename_all="lowercase")]
pub enum RpcDiagnosticSeverity { Error, Warning }
#[derive(Clone, Debug, PartialEq, Deserialize)]
#[serde(deny_unknown_fields)]
pub struct RpcDiagnosticData { pub schema: RpcDiagnosticDataSchema, pub diagnostics: Vec<RpcDiagnostic> }
#[derive(Clone, Debug, PartialEq, Deserialize)]
pub enum RpcDiagnosticDataSchema { #[serde(rename="semaprax.image-agent-application-error-data.v1")] V1 }
#[derive(Clone, Debug, PartialEq, Deserialize)]
#[serde(deny_unknown_fields)]
pub struct TypedRpcError { pub code: i64, pub message: String, #[serde(default)] pub data: Option<RpcDiagnosticData> }
#[derive(Clone, Debug, PartialEq)]
pub enum TypedDecodeError { Invalid(String), Rpc(TypedRpcError) }
impl std::fmt::Display for TypedDecodeError {
    fn fmt(&self, formatter:&mut std::fmt::Formatter<'_>)->std::fmt::Result { match self { Self::Invalid(error)=>formatter.write_str(error), Self::Rpc(error)=>write!(formatter,"RPC {}: {}",error.code,error.message) } }
}
impl std::error::Error for TypedDecodeError {}
fn metadata() -> Result<Value,String> { serde_json::from_str(METADATA).map_err(|e|e.to_string()) }
fn check(value:&Value, schema:&Value, meta:&Value, depth:usize)->Result<(),String> {
    check_bounded(value,schema,meta,depth,&mut 1_048_576usize)
}
fn check_bounded(value:&Value, schema:&Value, meta:&Value, depth:usize, work:&mut usize)->Result<(),String> {
    *work=work.checked_sub(1).ok_or("shape work exceeded")?;
    if depth>128 { return Err("shape depth exceeded".into()); }
    if let Some(reference)=schema["$ref"].as_str() {
        if let Some(document)=meta["documents"].get(reference) { return check_bounded(value,document,meta,depth+1,work); }
        if meta["unbundled"].as_array().is_some_and(|items|items.iter().any(|item|item==reference)) && value["schema"]==reference.strip_prefix("urn:").unwrap_or(reference) { return Ok(()); }
        return Err("unresolved or foreign payload schema".into());
    }
    for alternative in ["oneOf","anyOf"] {
        if let Some(branches)=schema[alternative].as_array() {
            let mut count=0;
            for branch in branches {
                match check_bounded(value,branch,meta,depth+1,work) {
                    Ok(())=>count+=1,
                    Err(error) if error=="shape work exceeded" || error=="shape depth exceeded"=>return Err(error),
                    Err(_)=>{},
                }
            }
            if (alternative=="oneOf" && count!=1) || (alternative=="anyOf" && count==0) { return Err("shape alternatives mismatch".into()); }
        }
    }
    if schema.get("const").is_some_and(|expected|value!=expected) { return Err("constant mismatch".into()); }
    if schema["enum"].as_array().is_some_and(|values|!values.contains(value)) { return Err("choice mismatch".into()); }
    if schema["not"]["enum"].as_array().is_some_and(|values|values.contains(value)) { return Err("excluded choice".into()); }
    match schema["type"].as_str() {
        Some("null") if !value.is_null()=>return Err("null required".into()),
        Some("boolean") if !value.is_boolean()=>return Err("boolean required".into()),
        Some("integer")=>{
            let n=value.as_i64().map(i128::from).or_else(||value.as_u64().map(i128::from)).ok_or("integer required")?;
            let min=schema["minimum"].as_i64().map(i128::from).or_else(||schema["minimum"].as_u64().map(i128::from)).unwrap_or(i64::MIN as i128);
            let max=schema["maximum"].as_u64().map(i128::from).or_else(||schema["maximum"].as_i64().map(i128::from)).unwrap_or(u64::MAX as i128);
            if n<min || n>max { return Err("integer bound".into()); }
        },
        Some("string")=>{
            let text=value.as_str().ok_or("string required")?;
            let chars=text.chars().count() as u64;
            if chars<schema["minLength"].as_u64().unwrap_or(0) || chars>schema["maxLength"].as_u64().unwrap_or(u64::MAX) || text.len() as u64>schema["x-max-utf8-bytes"].as_u64().unwrap_or(u64::MAX) { return Err("string bound".into()); }
            if let Some(pattern)=schema["pattern"].as_str() {
                if pattern=="^sha256:[0-9a-f]{64}$" {
                    if text.len()!=71 || !text.starts_with("sha256:") || !text.as_bytes()[7..].iter().all(|c|c.is_ascii_digit() || (b'a'..=b'f').contains(c)) { return Err("digest format".into()); }
                } else if pattern=="^[^\\u0000-\\u001f\\u007f-\\u009f]+$" {
                    if text.is_empty() || text.chars().any(char::is_control) { return Err("control character".into()); }
                } else if pattern=="^[A-Za-z_][A-Za-z0-9_]*$" {
                    let mut bytes=text.bytes();
                    if !bytes.next().is_some_and(|byte|byte.is_ascii_alphabetic() || byte==b'_') || !bytes.all(|byte|byte.is_ascii_alphanumeric() || byte==b'_') { return Err("identifier format".into()); }
                } else if pattern=="^[A-Za-z0-9_.:-]+$" {
                    if text.is_empty() || !text.bytes().all(|byte|byte.is_ascii_alphanumeric() || matches!(byte,b'_'|b'.'|b':'|b'-')) { return Err("binding format".into()); }
                } else if pattern=="^[a-z0-9._-]+$" {
                    if text.is_empty() || !text.bytes().all(|byte|byte.is_ascii_lowercase() || byte.is_ascii_digit() || matches!(byte,b'.'|b'_'|b'-')) { return Err("stable identity format".into()); }
                } else if pattern=="^[0-9a-f]{8}$" {
                    if text.len()!=8 || !text.bytes().all(|byte|byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte)) { return Err("32-bit hexadecimal format".into()); }
                } else if pattern=="^[0-9a-f]{16}$" {
                    if text.len()!=16 || !text.bytes().all(|byte|byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte)) { return Err("64-bit hexadecimal format".into()); }
                } else { return Err("unsupported client pattern; use compiler schema validator".into()); }
            }
        },
        Some("object")=>{
            let object=value.as_object().ok_or("object required")?;
            if schema["required"].as_array().is_some_and(|names|names.iter().any(|name|!object.contains_key(name.as_str().unwrap_or("")))) { return Err("missing field".into()); }
            // Reject a foreign discriminant before walking a recursive body/argument.
            if let Some(properties)=schema["properties"].as_object() {
                for (name,field) in properties {
                    if field.get("const").is_some() {
                        if let Some(item)=object.get(name) { check_bounded(item,field,meta,depth+1,work)?; }
                    }
                }
            }
            for(name,item)in object {
                if let Some(field)=schema["properties"].get(name) { check_bounded(item,field,meta,depth+1,work)?; }
                else if schema["additionalProperties"]==false { return Err(format!("unknown field {name}")); }
            }
        },
        Some("array")=>{
            let values=value.as_array().ok_or("array required")?;
            if (values.len() as u64)<schema["minItems"].as_u64().unwrap_or(0) || (values.len() as u64)>schema["maxItems"].as_u64().unwrap_or(u64::MAX) { return Err("array bound".into()); }
            if let Some(items)=schema.get("items") { for value in values { check_bounded(value,items,meta,depth+1,work)?; } }
            if schema["uniqueItems"]==true {
                let mut seen=std::collections::BTreeSet::new();
                for value in values {
                    *work=work.checked_sub(1).ok_or("shape work exceeded")?;
                    let text=value.as_str().ok_or("unique string items required")?;
                    if !seen.insert(text) { return Err("unique string items required".into()); }
                }
            }
        }, _=>{},
    }
    Ok(())
}
fn id_check(id:&RpcId)->Result<(),String> {
    if let RpcId::Text(text)=id {
        if text.is_empty() || text.len()>128 || text.chars().any(char::is_control) { return Err("request id bound/format".into()); }
    }
    Ok(())
}
pub fn request(id:RpcId,method:&str,params:Value)->Result<String,String> {
    id_check(&id)?;
    let meta=metadata()?;
    let entry=meta["methods"].get(method).ok_or("method unavailable")?;
    let mut spec=entry["params"].clone();
    for field in spec["properties"].as_object_mut().ok_or("invalid embedded parameter schema")?.values_mut() { field.as_object_mut().ok_or("invalid parameter")?.remove("$ref"); }
    check(&params,&spec,&meta,0)?;
    let line=json!({"jsonrpc":"2.0","id":id,"method":method,"params":params}).to_string();
    if line.len()>65536 { return Err("request byte bound".into()); }
    Ok(line+"\n")
}
fn validate_rpc_error_value(error:&Value,response_id:&Value,expected_id:&Value)->Result<(),String> {
    let fields=error.as_object().ok_or("RPC error must be an object")?;
    if fields.len()<2 || fields.len()>3 || !fields.contains_key("code") || !fields.contains_key("message") || fields.keys().any(|key|!matches!(key.as_str(),"code"|"message"|"data")) { return Err("RPC error fields are not closed".into()); }
    if fields.get("data").is_some_and(Value::is_null) { return Err("RPC diagnostic data must be omitted or an object".into()); }
    if let Some(data)=fields.get("data") {
        let data=data.as_object().ok_or("RPC diagnostic data must be an object")?;
        if data.len()!=2 || !data.contains_key("schema") || !data.contains_key("diagnostics") { return Err("RPC diagnostic data fields are not closed".into()); }
        let rows=data["diagnostics"].as_array().ok_or("RPC diagnostics must be an array")?;
        if rows.is_empty() { return Err("RPC diagnostics must not be empty".into()); }
        if fields["code"]!=-32000 || response_id.is_null() || response_id!=expected_id { return Err("application error data requires correlated -32000 response".into()); }
        for row in rows {
            let row=row.as_object().ok_or("RPC diagnostic must be an object")?;
            if row.len()!=6 || ["code","severity","message","path","location","help"].iter().any(|key|!row.contains_key(*key)) { return Err("RPC diagnostic fields are not closed".into()); }
            if let Some(location)=row["location"].as_object() {
                if location.len()!=4 || ["line","column","start","end"].iter().any(|key|!location.contains_key(*key)) { return Err("RPC diagnostic location fields are not closed".into()); }
            } else if !row["location"].is_null() { return Err("RPC diagnostic location must be null or an object".into()); }
        }
    }
    Ok(())
}
pub fn decode(line:&str,method:&str,id:&RpcId)->Result<ResultEnvelope,String> {
    id_check(id)?;
    if line.len()>1048576 { return Err("response byte bound".into()); }
    let meta=metadata()?;
    let entry=meta["methods"].get(method).ok_or("method unavailable")?;
    let value:Value=serde_json::from_str(line).map_err(|e|e.to_string())?;
    if value["jsonrpc"]!="2.0" { return Err("response envelope".into()); }
    let expected_id=serde_json::to_value(id).map_err(|e|e.to_string())?;
    if let Some(error)=value.get("error") {
        if value.as_object().is_none_or(|fields|fields.len()!=3 || !fields.contains_key("id")) { return Err("error envelope".into()); }
        if !value["id"].is_null() && value["id"]!=expected_id { return Err("response id mismatch".into()); }
        validate_rpc_error_value(error,&value["id"],&expected_id)?;
        let error:TypedRpcError=serde_json::from_value(error.clone()).map_err(|e|e.to_string())?;
        return Err(format!("RPC {}: {}",error.code,error.message));
    }
    if value.as_object().is_none_or(|fields|fields.len()!=3 || !fields.contains_key("result")) || value["id"]!=expected_id { return Err("response id/envelope mismatch".into()); }
    let result:ResultEnvelope=serde_json::from_value(value["result"].clone()).map_err(|e|e.to_string())?;
    if result.schema!=RESULT_SCHEMA || result.protocol!=PROTOCOL { return Err("protocol mismatch".into()); }
    let digest=json!({"type":"string","pattern":"^sha256:[0-9a-f]{64}$"});
    let mut work=1_048_576usize;
    check_bounded(&json!(result.image_revision),&digest,&meta,0,&mut work)?;
    check_bounded(&json!(result.project_revision),&digest,&meta,0,&mut work)?;
    check_bounded(&result.payload,&entry["payload"],&meta,0,&mut work)?;
    Ok(result)
}
/// Decode a selected response while preserving structured application diagnostics.
pub fn decode_typed(line:&str,method:&str,id:&RpcId)->Result<ResultEnvelope,TypedDecodeError> {
    id_check(id).map_err(TypedDecodeError::Invalid)?;
    if line.len()>1048576 { return Err(TypedDecodeError::Invalid("response byte bound".into())); }
    let meta=metadata().map_err(TypedDecodeError::Invalid)?;
    if meta["methods"].get(method).is_none() { return Err(TypedDecodeError::Invalid("method unavailable".into())); }
    let value:Value=serde_json::from_str(line).map_err(|error|TypedDecodeError::Invalid(error.to_string()))?;
    if value["jsonrpc"]!="2.0" { return Err(TypedDecodeError::Invalid("response envelope".into())); }
    if let Some(error)=value.get("error") {
        if value.as_object().is_none_or(|fields|fields.len()!=3 || !fields.contains_key("id")) { return Err(TypedDecodeError::Invalid("error envelope".into())); }
        let expected_id=serde_json::to_value(id).map_err(|error|TypedDecodeError::Invalid(error.to_string()))?;
        if !value["id"].is_null() && value["id"]!=expected_id { return Err(TypedDecodeError::Invalid("response id mismatch".into())); }
        validate_rpc_error_value(error,&value["id"],&expected_id).map_err(TypedDecodeError::Invalid)?;
        let error:TypedRpcError=serde_json::from_value(error.clone()).map_err(|error|TypedDecodeError::Invalid(error.to_string()))?;
        return Err(TypedDecodeError::Rpc(error));
    }
    decode(line,method,id).map_err(TypedDecodeError::Invalid)
}
